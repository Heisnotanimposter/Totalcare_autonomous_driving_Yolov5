\section{1. Split-Screen Layout}

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item \textbf{Total Composite Resolution}: \texttt{1360 $\times$ 632} (\texttt{Left:\allowbreak{} 720 $\times$ 540} | \texttt{Right:\allowbreak{} 640 $\times$ 540} | \texttt{Bottom Telemetry HUD:\allowbreak{} 1360 $\times$ 92}) at \texttt{10 Hz} (matching KITTI native frame rate).
\end{itemize}

\subsection{Left Panel --- Observed World}
Displays \textbf{only} direct camera-frame observations:

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item Original KITTI \texttt{image\_\allowbreak{}02} RGB frame
\item 2D Bounding Boxes color-coded by class (\texttt{Car}, \texttt{Pedestrian}, \texttt{Cyclist})
\item Class Label, sequence-scoped Track ID (\texttt{\#id}), and Detection Confidence (\texttt{0.\allowbreak{}00--1.\allowbreak{}00})
\end{itemize}

\subsection{Right Panel --- Inferred World}
Structured Ego-centric Bird's-Eye-View (\texttt{X:\allowbreak{} [-\allowbreak{}18m,\allowbreak{} +18m]}, \texttt{Y:\allowbreak{} [-\allowbreak{}10m,\allowbreak{} +52m]}):

\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{Ego Vehicle}: Yellow triangle anchored at \texttt{(0,\allowbreak{} 0)}
\item \textbf{Surrounding Objects}: Class-colored BEV footprints with ID and forward distance (\texttt{m})
\item \textbf{Relative Position}: Metric range rings (\texttt{10m}--\texttt{50m}) in Ego coordinates
\item \textbf{Movement Direction \& Calibrated Velocity}: Arrow \& \texttt{m/\allowbreak{}s} label shown \textbf{only} when \texttt{P2} calibration is active and track hits \texttt{>= 3}
\item \textbf{Predicted Trajectory}: \texttt{+0.\allowbreak{}5s}, \texttt{+1.\allowbreak{}0s}, \texttt{+1.\allowbreak{}5s}, \texttt{+2.\allowbreak{}0s} Kalman forecast nodes
\item \textbf{Ego Planned Path}: Forward corridor (\texttt{$\pm$1.\allowbreak{}6m} half-width, \texttt{0--35m} horizon)
\item \textbf{Road Structure}: Nominal lane boundaries aligned to Ego corridor
\item \textbf{Occupancy / Obstacles}: Solid occupied regions vs angular shadow cones
\item \textbf{Collision Risk}: Conflict zone circles (\texttt{CONFLICT +t\_\allowbreak{}s}) and risk highlight when \texttt{Risk Indicator >= 0.\allowbreak{}45}
\item \textbf{Uncertainty Visualization}: \texttt{1$\sigma$} covariance ellipses around each future horizon node + HUD breakdown (\texttt{Det}, \texttt{Depth $\pm$m}, \texttt{Track}, \texttt{Pred +2s $\pm$m})
\item \textbf{Temporal State Separation}: Fading \texttt{Past World} trails (exponentially decayed weight) vs \texttt{Present World} boxes vs \texttt{Future World} probabilistic cones
\item \textbf{Spatial Limits}: Rear area (\texttt{Y < 0}) and outside camera horizontal FOV explicitly hatched and labeled \texttt{UNKNOWN (Outside Front Camera FOV)}
\end{enumerate}
