\section{1. Research Hypotheses}

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item \textbf{H1 (Sequence-Disjoint Integrity \& Masking):} Sequence-disjoint splitting combined with \texttt{DontCare} + distractor (\texttt{Van}, \texttt{Person}, \texttt{Person\_\allowbreak{}sitting}) masking prevents temporal overfitting and spurious false-negative gradients compared to unmasked or frame-random splits.
\item \textbf{H2 (Temporal Continuity):} Associating 2D detections with ByteTrack (\texttt{src/\allowbreak{}track\_\allowbreak{}utils.\allowbreak{}py}) and filtering ego-centric trajectories via a kinematic Kalman filter (\texttt{src/\allowbreak{}world\_\allowbreak{}model.\allowbreak{}py}) stabilizes noisy monocular ground-plane depth estimates and preserves object identity across brief occlusions.
\item \textbf{H3 (Uncertainty-Aware Prediction \& Risk):} Separating detection, depth, tracking, and motion prediction uncertainties yields calibrated multi-horizon trajectory bounds (\texttt{t + 0.\allowbreak{}5s} to \texttt{+2.\allowbreak{}0s}) and more consistent risk indicators than naive linear bounding-box extrapolation.
\end{itemize}

\section{2. Baselines \& Ablation Studies}

\subsection{Baselines}
\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{B0 --- Frame-Independent Detection (\texttt{src/\allowbreak{}detect.\allowbreak{}py})}: Per-frame 2D bounding boxes with no temporal identity or ego-centric spatial projection.
\item \textbf{B1 --- Detector + 2D Tracker (\texttt{YOLOv8s + ByteTrack}, \texttt{src/\allowbreak{}track\_\allowbreak{}utils.\allowbreak{}py})}: 2D image-plane tracking with \texttt{px/\allowbreak{}sec} velocity only.
\item \textbf{B2 --- Temporal Ego-centric World Model (\texttt{src/\allowbreak{}world\_\allowbreak{}model.\allowbreak{}py})}: Calibrated \texttt{P2} ground-plane projection + 4D kinematic Kalman filter + \texttt{Observed/\allowbreak{}Inferred/\allowbreak{}Occluded/\allowbreak{}Unknown} state machine + Safety Shield.
\end{enumerate}

\section{4. Experimental Execution Conditions}

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item \textbf{Execution Environment}: Kaggle CLI GPU (\texttt{NvidiaTeslaT4}, auto-DP \texttt{2x T4} supported) and/or Google Colab CLI (local CPU lacks PyTorch/CUDA).
\item \textbf{Target Classes}: \texttt{Car}, \texttt{Pedestrian}, \texttt{Cyclist} (\texttt{nc = 3}).
\item \textbf{Training Hyperparameters (v1 Baseline)}: \texttt{yolov8s.\allowbreak{}pt}, \texttt{epochs=30}, \texttt{imgsz=640}, \texttt{batch=16}, \texttt{patience=10}, \texttt{device=0}.
\item \textbf{Training Hyperparameters (v2 Improved)}: \texttt{yolo11m.\allowbreak{}pt}, \texttt{epochs=40} (early-stopped at 24, best epoch 12), \texttt{imgsz=960}, \texttt{batch=16} (\texttt{8 x 2 GPUs}), \texttt{patience=12}, \texttt{cos\_\allowbreak{}lr=True}, \texttt{close\_\allowbreak{}mosaic=10}, \texttt{mixup=0.\allowbreak{}1}, \texttt{oversample\_\allowbreak{}rare\_\allowbreak{}classes=True} (\texttt{Cyclist x4}, \texttt{Pedestrian x2}, \texttt{6,\allowbreak{}974} weighted train entries; \texttt{val}/\texttt{test} untouched).
\item \textbf{Tracking, Masking \& World Model Params}: \texttt{ByteTrack} @ \texttt{imgsz=960}, \texttt{conf=0.\allowbreak{}25}, \texttt{mask\_\allowbreak{}model=mobile\_\allowbreak{}sam.\allowbreak{}pt}, \texttt{fps=10.\allowbreak{}0}, \texttt{camera\_\allowbreak{}height\_\allowbreak{}m=1.\allowbreak{}65}, \texttt{max\_\allowbreak{}coast\_\allowbreak{}sec=1.\allowbreak{}0}, \texttt{horizons=(0.\allowbreak{}5,\allowbreak{} 1.\allowbreak{}0,\allowbreak{} 1.\allowbreak{}5,\allowbreak{} 2.\allowbreak{}0)}, \texttt{CollisionRiskSmoother} (\texttt{alpha\_\allowbreak{}up=0.\allowbreak{}36}, \texttt{alpha\_\allowbreak{}down=0.\allowbreak{}22}, \texttt{max\_\allowbreak{}rise\_\allowbreak{}per\_\allowbreak{}sec=1.\allowbreak{}10}, \texttt{max\_\allowbreak{}fall\_\allowbreak{}per\_\allowbreak{}sec=0.\allowbreak{}85}, \texttt{hysteresis\_\allowbreak{}margin=0.\allowbreak{}04}).
