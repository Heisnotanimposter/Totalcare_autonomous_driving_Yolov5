\section{1. 연구 가설 (Research Hypotheses)}

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item \textbf{H1 (시퀀스 비독립 무결성 및 마스킹)}: 시퀀스 비독립 분할과 \texttt{DontCare} + Distractor(\texttt{Van}, \texttt{Person}, \texttt{Person\_\allowbreak{}sitting}) 마스킹을 결합하면 비마스킹 또는 프레임 단위 무작위 분할 대비 시계열 과적합과 잘못된 False-Negative 경사를 방지할 수 있습니다.
\item \textbf{H2 (시계열 연속성)}: 2D 검출 결과를 ByteTrack(\texttt{src/\allowbreak{}track\_\allowbreak{}utils.\allowbreak{}py})으로 연결하고 운동학 칼만 필터(\texttt{src/\allowbreak{}world\_\allowbreak{}model.\allowbreak{}py})로 자차 중심 궤적을 필터링하면 단안 지면 깊이 추정의 잡음을 안정화하고 짧은 가려짐(Occlusion) 구간에서도 객체 정체성을 유지할 수 있습니다.
\item \textbf{H3 (불확실성 기반 예측 및 위험도 평가)}: 검출, 깊이, 추적, 운동 예측의 불확실성을 분리 모델링하면 단순 선형 BBox 외삽 대비 보정된 다중 시점 궤적 범위(\texttt{t + 0.\allowbreak{}5s} \textasciitilde{} \texttt{+2.\allowbreak{}0s})와 일관된 위험도 지표를 산출할 수 있습니다.
\end{itemize}

\section{2. 베이스라인 및 절제 연구(Ablation Studies)}

\subsection{베이스라인 모델}
\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{B0 --- 프레임 독립 검출 (\texttt{src/\allowbreak{}detect.\allowbreak{}py})}: 시계열 ID나 자차 중심 공간 투영이 없는 프레임별 2D 바운딩 박스 검출.
\item \textbf{B1 --- 검출기 + 2D 추적기 (\texttt{YOLOv8s + ByteTrack}, \texttt{src/\allowbreak{}track\_\allowbreak{}utils.\allowbreak{}py})}: \texttt{px/\allowbreak{}sec} 속도만 사용하는 2D 이미지 평면 추적.
\item \textbf{B2 --- 시계열 자차 중심 World Model (\texttt{src/\allowbreak{}world\_\allowbreak{}model.\allowbreak{}py})}: 보정된 \texttt{P2} 지면 투영 + 4D 운동학 칼만 필터 + \texttt{Observed/\allowbreak{}Inferred/\allowbreak{}Occluded/\allowbreak{}Unknown} 상태 머신 + Safety Shield.
\end{enumerate}
