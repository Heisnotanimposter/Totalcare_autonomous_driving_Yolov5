\subsection{2. Training Progression \& Evolution (\texttt{v1 Baseline $\rightarrow$ v2 Improved $\rightarrow$ v3 BEV Fixed})}

\subsubsection{2.1 Training Setup \& Architecture Evolution}
\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{\texttt{v1} Baseline (\texttt{YOLOv8s @ 640px})}: \texttt{epochs=30}, \texttt{batch=16}, \texttt{129 layers /\allowbreak{} 11.\allowbreak{}14M params /\allowbreak{} 28.\allowbreak{}6 GFLOPs}.
\item \textbf{\texttt{v2} Improved (\texttt{YOLO11m @ 960px} + Rare-Class Oversampling + \texttt{MobileSAM})}:
\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item Upgraded input resolution from \texttt{640 $\rightarrow$ 960} with \texttt{C3k2 + C2PSA} spatial-attention \texttt{YOLO11m} (\texttt{231 layers /\allowbreak{} 20.\allowbreak{}06M params /\allowbreak{} 68.\allowbreak{}3 GFLOPs}).
\item Applied rare-class oversampling (\texttt{6,\allowbreak{}974} weighted train entries; Val/Test untouched), \texttt{cos\_\allowbreak{}lr=True}, \texttt{mixup=0.\allowbreak{}1}, \texttt{close\_\allowbreak{}mosaic=10}, \texttt{patience=12} on \textbf{\texttt{2$\times$ Tesla T4}} DDP (early-stopped at \texttt{Epoch 24}, Best = \textbf{\texttt{Epoch 12}}, \textasciitilde{}108 min).
\item Coupled \texttt{mobile\_\allowbreak{}sam.\allowbreak{}pt} (\texttt{38.\allowbreak{}8 MB}) Box-Prompt segmentation for pixel masks and ground-contact points (\texttt{mask\_\allowbreak{}bottom\_\allowbreak{}xy}).
\end{itemize}
\item \textbf{\texttt{v3} BEV Visual \& False-Alarm Fix (\texttt{world\_\allowbreak{}model.\allowbreak{}py}, \texttt{world\_\allowbreak{}visualizer.\allowbreak{}py})}:
\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item Fixed >2x lateral distortion of longitudinal uncertainty ellipses in the BEV panel via Jacobian projection $J=\text{diag}(s_x,-s_y)$, lateral covariance decoupling, approach-speed gating (\texttt{approach\_\allowbreak{}w}), and per-track EMA smoothing.
\end{itemize}
\end{enumerate}

\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{\texttt{YOLO11m} (\texttt{231 layers}, \texttt{20.\allowbreak{}06M params}, \texttt{68.\allowbreak{}3 GFLOPs})}: Backbone (\texttt{Conv} Stem $\rightarrow$ \texttt{C3k2} CSP Bottleneck $\rightarrow$ \texttt{SPPF} $\rightarrow$ \texttt{C2PSA} Spatial Attention), Neck (\texttt{PAN-\allowbreak{}FPN}), Head (\texttt{Decoupled Anchor-\allowbreak{}Free Detect} + \texttt{DFL} + \texttt{CIoU} + \texttt{BCE}).
\item \textbf{\texttt{MobileSAM} (\texttt{38.\allowbreak{}8 MB})}: Knowledge-Distilled \texttt{TinyViT} Image Encoder (\texttt{MBConv} + \texttt{Windowed Self-\allowbreak{}Attention}), Box Prompt Positional Encoder, Two-Way Cross-Attention Mask Decoder + \texttt{4$\times$} Transposed Conv.
\item \textbf{\texttt{ByteTrack}}: 8D Constant-Velocity BBox Kalman Filter + Two-Stage Hungarian Bipartite Matching (\texttt{High-\allowbreak{}Conf} + \texttt{Low-\allowbreak{}Conf} occlusion recovery).
\item \textbf{Temporal World Model \& Safety Shield}: \texttt{P2} inverse-variance depth fusion + OXTS \texttt{SE(2)} compensation + 4D World-ENU Kalman filter + Jacobian BEV covariance ellipse projection + $C^1$ hazard potential \& \texttt{CollisionRiskSmoother} slew-rate/hysteresis smoothing.
\item \textbf{\texttt{SegFormer-\allowbreak{}B2} (\texttt{nvidia/\allowbreak{}segformer-\allowbreak{}b2-\allowbreak{}finetuned-\allowbreak{}cityscapes-\allowbreak{}1024-\allowbreak{}1024}, added in v4)}: Hierarchical MiT-B2 Mix-Transformer encoder + All-MLP decoder (\texttt{Non-\allowbreak{}Road} / \texttt{Other Road} / \texttt{Ego-\allowbreak{}Lane} 3-class head) + BEV IPM warp \& HLS+Sobel quadratic polynomial EMA lane tracker.
\end{enumerate}

\subsection{3. Multi-Object Tracking Layer: \texttt{ByteTrack} (\texttt{src/\allowbreak{}track\_\allowbreak{}utils.\allowbreak{}py})}

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item \textbf{State Estimation Filter (Linear Kalman Filter)}: 8D constant-velocity image-plane state $\mathbf{x} = [c_x, c_y, a, h, \dot{c}_x, \dot{c}_y, \dot{a}, \dot{h}]^T$ (center, aspect ratio, height, and velocities).
\item \textbf{Two-Stage Hungarian Association}:
\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{High-Confidence Matching (\texttt{conf >= 0.\allowbreak{}5})}: Applies \texttt{linear\_\allowbreak{}sum\_\allowbreak{}assignment} on the IoU cost matrix between active tracks and high-confidence boxes.
\item \textbf{Low-Confidence Recovery (\texttt{0.\allowbreak{}1 <= conf < 0.\allowbreak{}5})}: Re-associates lower-confidence detections caused by partial occlusion or backlight to unmatched tracks, reducing ID switches.
\end{enumerate}
\item \textbf{Sequence Isolation Audit (\texttt{compute\_\allowbreak{}track\_\allowbreak{}trajectories})}: Resets tracker state at sequence boundaries, indexes by composite \texttt{(sequence\_\allowbreak{}id,\allowbreak{} track\_\allowbreak{}id)}, and logs any frame-order, duplicate, or class-flip anomalies.
\end{itemize}

\subsection{5. Temporal Ego-centric World Model \& Safety Shield Layer (\texttt{src/\allowbreak{}world\_\allowbreak{}model.\allowbreak{}py}, \texttt{src/\allowbreak{}world\_\allowbreak{}visualizer.\allowbreak{}py})}

Differentiable Kalman-based backend unifying perception outputs into metric physics and temporal continuity:


\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{P2 Inverse-Variance Depth Fusion Layer (\texttt{KittiCalibration.\allowbreak{}ground\_\allowbreak{}plane\_\allowbreak{}bev\_\allowbreak{}fusion})}:
\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item Combines flat-ground projection ($Y_{\text{ground}} = f_y h_{\text{cam}} / (v - c_y)$) anchored at \texttt{mask\_\allowbreak{}bottom\_\allowbreak{}xy} (\texttt{mask\_\allowbreak{}fill\_\allowbreak{}ratio >= 0.\allowbreak{}25}) with pitch-invariant class-height depth ($Y_{\text{height}} = f_y H_c / h_{\text{px}}$) via $Y_{\text{fused}} = \frac{w_g Y_g + w_h Y_h}{w_g + w_h}$.
\end{itemize}
\item \textbf{OXTS \texttt{SE(2)} Ego-Motion Compensation \& CTRV Layer (\texttt{EgoPose2D}, \texttt{KittiOxtsOdometry})}:
\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item Mercator-projected \texttt{World-\allowbreak{}ENU} transform removes Ego translation and yaw ($\psi, v_f, w_z$) to eliminate false velocities on stationary objects (\texttt{round-\allowbreak{}trip error < 4e-\allowbreak{}14 m}).
\end{itemize}
\item \textbf{4D World-ENU Linear Kalman Filter (\texttt{ObjectWorldTrack})}:
\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item State $\mathbf{x}_w = [X_w, Y_w, v_{X,w}, v_{Y,w}]^T$, white-noise acceleration process noise ($\sigma_a = 1.1\text{ m/s}^2$), rotated measurement covariance $R_w$ scaled inversely by confidence and mask fill ratio, projected to BEV pixels via Jacobian $J=\text{diag}(s_x,-s_y)$.
\end{itemize}
\item \textbf{$C^1$ Continuous Hazard Potential + \texttt{CollisionRiskSmoother} Layer}:
\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item Lateral-covariance (\texttt{lateral\_\allowbreak{}uncertainty\_\allowbreak{}m}) corridor intrusion, approach-speed gating (\texttt{approach\_\allowbreak{}w}), non-road attenuation (\texttt{0.\allowbreak{}65} in \texttt{v4}), calibrated TTC, and probabilistic soft-union $1 - \prod_i (1 - r_i)$.
\item Asymmetric EMA (\texttt{alpha\_\allowbreak{}up=0.\allowbreak{}36}, \texttt{alpha\_\allowbreak{}down=0.\allowbreak{}22}), slew-rate clamping (\texttt{1.\allowbreak{}10/\allowbreak{}s}, \texttt{-\allowbreak{}0.\allowbreak{}85/\allowbreak{}s}), per-track EMA (\texttt{per\_\allowbreak{}track\_\allowbreak{}ema}), low-TTC debouncing (\texttt{low\_\allowbreak{}ttc\_\allowbreak{}streak}), and hysteresis (\texttt{0.\allowbreak{}04}) eliminate \texttt{>1.\allowbreak{}5/\allowbreak{}s} spikes by \texttt{100\%}.
\end{itemize}
\end{enumerate}

\section{Architecture: Camera-Centric Temporal Ego-centric World Model}

\subsection{1. System Overview}

This project implements a 7-stage pipeline transforming a single forward-facing camera stream into a structured \textbf{Temporal Ego-centric World Model}, multi-horizon trajectory predictions, risk indicators, and an advisory Safety Shield output:


\begin{quote}\begin{footnotesize}\begin{verbatim}
[Stage 1: Input Video (KITTI 10 Hz)]
        |
        v
[Stage 2: Visual Perception (src/kitti_prep.py & YOLO11m @ 960px + MobileSAM)]
        |  2D BBox, Class (Car, Pedestrian, Cyclist), Detection Confidence
        |  Rare-Class Train Oversampling (Cyclist x4, Pedestrian x2) + Instance Polygons
        v
[Stage 3: Temporal Understanding (src/track_utils.py, ByteTrack @ 960px + MobileSAM)]
        |  Sequence-scoped Track ID, Frame ID, Timestamp, mask_area_px, mask_fill_ratio, mask_bottom_xy
        v
[Stage 4: Ego-centric World Model (src/world_model.py)]
        |  P2 Inverse-Variance Depth Fusion (Flat Ground + Class-Height Prior + Mask Bottom Contact)
        |  OXTS IMU/GPS Ego-Motion Compensation (EgoPose2D: Ego <-> World ENU)
        |  State Machine: Past <-> Present <-> Future | Observed / Inferred / Occluded / Unknown
        v
[Stage 5: Future Prediction (src/world_model.py, World-ENU Kalman + CTRV Ego)]
        |  t + 0.5s, 1.0s, 1.5s, 2.0s Means + Confidence/Mask-Scaled 1-sigma Covariance Ellipses
        v

        |  Smooth Sigmoid Hazard Potential + Asymmetric EMA + Slew-Rate |dR/dt| Clamp + Calibrated TTC

           Hysteresis-Debounced Advisory Safety Shield + Split-Screen (Masks + BEV World)

\subsection{2. Coordinate Systems \& Transformations}

\subsubsection{2.1 KITTI Camera Coordinate Frame}
In KITTI Tracking (\texttt{calib/\allowbreak{}XXXX.\allowbreak{}txt}, \texttt{label\_\allowbreak{}02/\allowbreak{}XXXX.\allowbreak{}txt}):

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item $X_{\text{cam}}$: right ($+$)
\item $Y_{\text{cam}}$: down ($+$)
\item $Z_{\text{cam}}$: forward along optical axis ($+$)
\end{itemize}

\subsubsection{2.2 Ego-centric BEV Coordinate Frame}
Defined in \texttt{KittiCalibration} (\href{https://github.com/Heisnotanimposter/Totalcare\_autonomous\_driving\_Yolov5/blob/main/src/world\_model.py}{src/world\_model.py}):

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item $X_{\text{ego}}$: lateral right ($+$), meters
\item $Y_{\text{ego}}$: longitudinal forward ($+$), meters
\item $Z_{\text{ego}}$: height upward ($+$), meters
\end{itemize}

Under nominal level mounting ($h_{\text{cam}} = 1.65\text{ m}$, zero pitch/roll assumption):

\[X_{\text{ego}} = X_{\text{cam}}, \quad Y_{\text{ego}} = Z_{\text{cam}}, \quad Z_{\text{ego}} = h_{\text{cam}} - Y_{\text{cam}}\]

\subsubsection{2.3 Calibrated Inverse-Variance Depth Fusion \& Analytical Uncertainty}
Given KITTI intrinsics $P_2$ ($f_x, f_y, c_x, c_y$), contact point $(u_{\text{mid}}, v_{\text{bottom}})$ (using MobileSAM foreground bottom \texttt{mask\_\allowbreak{}bottom\_\allowbreak{}xy} when \texttt{mask\_\allowbreak{}fill\_\allowbreak{}ratio >= 0.\allowbreak{}25}, else the 2D box bottom-center), and class canonical height prior $H_c$ (\texttt{CLASS\_\allowbreak{}HEIGHT\_\allowbreak{}PRIORS\_\allowbreak{}M}: \texttt{Car=1.\allowbreak{}52m}, \texttt{Pedestrian=1.\allowbreak{}72m}, \texttt{Cyclist=1.\allowbreak{}68m}):

\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{Flat-Ground Projection ($Y_{\text{ground}}$)} with pitch uncertainty $\sigma_\theta = 0.012\text{ rad}$:
\end{enumerate}
\[Y_{\text{ground}} = \frac{f_y \cdot h_{\text{cam}}}{v_{\text{bottom}} - c_y}, \qquad \sigma_{\text{ground}} = \sqrt{\left(\frac{Y_{\text{ground}}^2}{f_y h_{\text{cam}}}\sigma_v\right)^2 + \left(\frac{Y_{\text{ground}}^2}{h_{\text{cam}}}\sigma_\theta\right)^2}\]
\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{Pitch-Invariant Height-Prior Depth ($Y_{\text{height}}$)} on non-truncated boxes:
\end{enumerate}
\[Y_{\text{height}} = \frac{f_y \cdot H_c}{h_{\text{px}}}, \qquad X_{\text{ego}} = \frac{(u_{\text{mid}} - c_x) \cdot Y_{\text{fused}}}{f_x}\]
\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{Inverse-Variance Fusion}:
\end{enumerate}
\[w_g = \sigma_{\text{ground}}^{-2}, \quad w_h = \sigma_{\text{height}}^{-2}, \qquad Y_{\text{fused}} = \frac{w_g Y_{\text{ground}} + w_h Y_{\text{height}}}{w_g + w_h}, \qquad \sigma_Y = (w_g + w_h)^{-1/2}\]
Measurement covariance $R_w$ in World-ENU Kalman filtering is further scaled inversely by detection confidence and MobileSAM mask fill quality (\texttt{conf}, \texttt{mask\_\allowbreak{}fill\_\allowbreak{}ratio}).


\textbf{Integrity Rules:}

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item If $P_2$ calibration is missing, \texttt{TemporalWorldModel} sets \texttt{calibrated = False} and \textbf{refuses} to compute metric speed (\texttt{m/\allowbreak{}s}) or Time-to-Collision (\texttt{TTC}).
\item Projected 3D locations are always categorized as \textbf{Inferred} (monocular ground-plane/height-fused estimate), never \textbf{Observed}.
\end{itemize}

\par\noindent\rule{\textwidth}{0.4pt}\par\medskip

\subsection{3. Ego-Motion Compensation \& Temporal World Model (\texttt{Past $\rightarrow$ Present $\rightarrow$ Future})}

Implemented in \texttt{EgoPose2D}, \texttt{KittiOxtsOdometry}, \texttt{TemporalWorldModel}, and \texttt{ObjectWorldTrack} (\href{https://github.com/Heisnotanimposter/Totalcare\_autonomous\_driving\_Yolov5/blob/main/src/world\_model.py}{src/world\_model.py}):


\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{OXTS Ego-Motion Compensation (\texttt{Ego $\leftrightarrow$ World ENU})}:
\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item \texttt{KittiOxtsOdometry} converts KITTI \texttt{oxts/\allowbreak{}<seq>.\allowbreak{}txt} \texttt{(lat,\allowbreak{} lon,\allowbreak{} yaw,\allowbreak{} vf,\allowbreak{} vl,\allowbreak{} wz)} into local metric \textbf{East-North (ENU)} coordinates via Mercator projection anchored at frame 0.
\item \texttt{EgoPose2D.\allowbreak{}ego\_\allowbreak{}to\_\allowbreak{}world\_\allowbreak{}xy} and \texttt{world\_\allowbreak{}to\_\allowbreak{}ego\_\allowbreak{}xy} transform instantaneous Ego-centric detections (\texttt{+X} right, \texttt{+Y} forward) into a stationary World-ENU frame (observed round-trip error \texttt{< 4e-\allowbreak{}14 m}).
\item Object velocities and 4D Kalman states \texttt{[X\_\allowbreak{}w,\allowbreak{} Y\_\allowbreak{}w,\allowbreak{} vX\_\allowbreak{}w,\allowbreak{} vY\_\allowbreak{}w]} are filtered in \textbf{World-ENU} (subtracting Ego translation and yaw rotation), while Ego's planned corridor uses a Constant Turn Rate and Velocity (\texttt{predict\_\allowbreak{}ctrv}) rollout from \texttt{vf} and \texttt{wz}.
\end{itemize}
\end{enumerate}

\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{Temporal State Transition (\texttt{Future\_\allowbreak{}prev $\rightarrow$ Present $\rightarrow$ Past})}:
\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item At timestamp $t_k$, each active track's World-frame 4D kinematic Kalman state and covariance $P$ are propagated forward by $\Delta t = t_k - t_{k-1}$ and projected back into Ego $t_k$'s BEV frame.
\item Tracks matched to current detections are updated with rotated measurement covariance $R_w = R_{\text{ego}\to w} \, \text{diag}(\sigma_X^2, \sigma_Y^2) \, R_{\text{ego}\to w}^T$ and assigned \texttt{observation\_\allowbreak{}state = "Observed"}.
\item Tracks not observed at $t_k$ within \texttt{max\_\allowbreak{}coast\_\allowbreak{}sec} ($1.0\text{ s}$) transition to \texttt{observation\_\allowbreak{}state = "Occluded"} with exponential confidence decay $\exp(-\Delta t_{\text{miss}} / 0.6)$.
\item Tracks missing beyond \texttt{max\_\allowbreak{}coast\_\allowbreak{}sec} archive into \texttt{past\_\allowbreak{}world.\allowbreak{}expired\_\allowbreak{}tracks} with their \texttt{last\_\allowbreak{}seen\_\allowbreak{}ts} and World-anchored position history.
\item Because \texttt{Past}, \texttt{Present}, and \texttt{Future} are indexed by timestamp relative to $t_k$ and maintained in World-ENU, their semantics remain invariant under Ego rotation or reversing.
\end{itemize}
\end{enumerate}

\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{Spatial Information States (\texttt{Observed /\allowbreak{} Inferred /\allowbreak{} Occluded /\allowbreak{} Unknown})}:
\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item \textbf{Observed}: Direct 2D camera measurements and active tracked detections inside the front camera FOV.
\item \textbf{Inferred}: Ground-plane BEV positions, kinematic velocities, Kalman future trajectories (\texttt{+0.\allowbreak{}5s} to \texttt{+2.\allowbreak{}0s}), and Risk Indicators.
\item \textbf{Occluded}: Angular shadow sectors ray-cast behind foreground objects (\texttt{compute\_\allowbreak{}occlusion\_\allowbreak{}sectors}) and temporarily coasting tracks.
\item \textbf{Unknown}: Regions behind Ego ($Y_{\text{ego}} < 0$) and outside the front camera horizontal FOV (\texttt{2 * atan2(W /\allowbreak{} 2,\allowbreak{} fx)}).
\end{itemize}
\end{enumerate}

\par\noindent\rule{\textwidth}{0.4pt}\par\medskip

\subsection{4. Safety Shield Policy \& Collision Detector Gradient Smoothing (Advisory Only)}

\texttt{assess\_\allowbreak{}risk\_\allowbreak{}and\_\allowbreak{}safety\_\allowbreak{}shield} combines:

\begin{enumerate}[leftmargin=1.8em,itemsep=2pt]
\item \textbf{$C^1$-Continuous Differentiable Risk Potential}: Per-object hazard scores use smooth \texttt{\_\allowbreak{}sigmoid} curves over true \textbf{lateral covariance} (\texttt{lateral\_\allowbreak{}uncertainty\_\allowbreak{}m}, \texttt{cov\_\allowbreak{}future\_\allowbreak{}ego[0][0]}) corridor intrusion bounds, approach-gated longitudinal distance (\texttt{approach\_\allowbreak{}w}, preventing stationary-vehicle and sidewalk false alarms), calibrated TTC, and compact earliest-horizon \texttt{conflict\_\allowbreak{}zones} (\texttt{[1.\allowbreak{}0,\allowbreak{} 2.\allowbreak{}2] m}, max 1 per track), aggregated via probabilistic soft-union $1 - \prod_i (1 - r_i)$.
\item \textbf{Aspect-Corrected BEV Uncertainty Ellipses (\texttt{BevCanvasMapper.\allowbreak{}cov\_\allowbreak{}to\_\allowbreak{}pixel\_\allowbreak{}ellipse})}: Projects World-ENU $2\times 2$ covariance into BEV pixel coordinates via Jacobian $J = \text{diag}(s_x, -s_y)$ so elongated longitudinal covariance is never stretched laterally across lanes, rendered as soft translucent horizon fills (\texttt{t = +1.\allowbreak{}0s,\allowbreak{} +2.\allowbreak{}0s}) and a single outer \texttt{+2.\allowbreak{}0s} contour.
\item \textbf{Gradient Smoothing (\texttt{CollisionRiskSmoother})}: Eliminates high-frequency detector/depth spikes across frames via asymmetric EMA (\texttt{alpha\_\allowbreak{}up=0.\allowbreak{}36}, \texttt{alpha\_\allowbreak{}down=0.\allowbreak{}22}), explicit slew-rate clamping (\texttt{max\_\allowbreak{}rise\_\allowbreak{}per\_\allowbreak{}sec=1.\allowbreak{}10/\allowbreak{}s}, \texttt{max\_\allowbreak{}fall\_\allowbreak{}per\_\allowbreak{}sec=0.\allowbreak{}85/\allowbreak{}s}), per-track risk EMA (\texttt{per\_\allowbreak{}track\_\allowbreak{}ema}), low-TTC debouncing, and hysteresis (\texttt{hysteresis\_\allowbreak{}margin=0.\allowbreak{}04}) across Safety Shield thresholds.
\end{enumerate}

Outputs are strictly \textbf{advisory}:

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item \texttt{Normal Driving}
\item \texttt{Caution}
\item \texttt{Speed Reduction}
\item \texttt{Braking Recommendation}
\item \texttt{Emergency Intervention}
\end{itemize}

\section{Experiment Plan: Camera-Centric Temporal Ego-centric World Model}

\subsection{1. Research Hypotheses}

\begin{itemize}[leftmargin=1.5em,itemsep=2pt]
\item \textbf{H1 (Sequence-Disjoint Integrity \& Masking):} Sequence-disjoint splitting combined with \texttt{DontCare} + distractor (\texttt{Van}, \texttt{Person}, \texttt{Person\_\allowbreak{}sitting}) masking prevents temporal overfitting and spurious false-negative gradients compared to unmasked or frame-random splits.
\item \textbf{H2 (Temporal Continuity):} Associating 2D detections with ByteTrack (\texttt{src/\allowbreak{}track\_\allowbreak{}utils.\allowbreak{}py}) and filtering ego-centric trajectories via a kinematic Kalman filter (\texttt{src/\allowbreak{}world\_\allowbreak{}model.\allowbreak{}py}) stabilizes noisy monocular ground-plane depth estimates and preserves object identity across brief occlusions.
\item \textbf{H3 (Uncertainty-Aware Prediction \& Risk):} Separating detection, depth, tracking, and motion prediction uncertainties yields calibrated multi-horizon trajectory bounds (\texttt{t + 0.\allowbreak{}5s} to \texttt{+2.\allowbreak{}0s}) and more consistent risk indicators than naive linear bounding-box extrapolation.
\end{itemize}
